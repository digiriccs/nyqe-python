  <div class="container">
    <div class="card">
      <div class="heading">SIP Growth Simulator</div>

      <label class="input-label" for="amount">Monthly Investment (₹)</label>
      <input id="amount" type="number" value="5000" />

      <label class="input-label" for="years">Investment Duration (Years)</label>
      <input id="years" type="number" value="10" />

      <label class="input-label" for="return">Expected Annual Return (%)</label>
      <input id="return" type="number" value="12" />

      <button class="btn-primary" onclick="simulateSIP()">Simulate SIP</button>

      <canvas id="sipChart"></canvas>

  <script>
    const ctx = document.getElementById('sipChart').getContext('2d');
    let sipChart;

    function calculateSIP(p, r, n) {
      r = r / 12 / 100;
      n = n * 12;
      return p * (((Math.pow(1 + r, n) - 1) / r) * (1 + r));
    }

    function simulateSIP() {
      const amount = parseFloat(document.getElementById('amount').value);
      const years = parseFloat(document.getElementById('years').value);
      const rate = parseFloat(document.getElementById('return').value);

      const invested = amount * years * 12;
      const maturity = calculateSIP(amount, rate, years);
      const gain = maturity - invested;

      const data = {
        labels: ['Invested', 'Gains'],
        datasets: [{
          label: 'SIP Result',
          data: [invested.toFixed(2), gain.toFixed(2)],
          backgroundColor: ['#003459', '#00A6ED'],
          borderWidth: 1
        }]
      };

      if (sipChart) sipChart.destroy();
      sipChart = new Chart(ctx, {
        type: 'doughnut',
        data: data,
        options: {
          plugins: {
            legend: { position: 'bottom' },
            tooltip: {
              callbacks: {
                label: function (ctx) {
                  return `₹ ${parseFloat(ctx.parsed).toLocaleString()}`;
                }
              }
            }
          }
        }
      });
    }

    // Initial load
    simulateSIP();
  </script>
