<script>
  const animations = [
    { id: 'lottie1', path: 'https://assets2.lottiefiles.com/packages/lf20_8wREpI.json' },
    { id: 'lottie2', path: 'https://assets2.lottiefiles.com/packages/lf20_vfnsb74e.json' },
    { id: 'lottie3', path: 'https://assets10.lottiefiles.com/packages/lf20_yxzrznz7.json' },
    { id: 'lottie4', path: 'https://assets7.lottiefiles.com/packages/lf20_totrpclr.json' }
  ];

  animations.forEach(({ id, path }) => {
    lottie.loadAnimation({
      container: document.getElementById(id),
      renderer: 'svg',
      loop: true,
      autoplay: true,
      path
    });
  });

  let currentSlide = 1;
  function nextSlide() {
    document.getElementById('slide' + currentSlide).style.display = 'none';
    currentSlide++;
    document.getElementById('slide' + currentSlide).style.display = 'block';
  }

  function endWalkthrough() {
    localStorage.setItem('walkthroughSeen', 'true');
    document.getElementById('walkthrough').style.display = 'none';
  }

  function startSite() {
    if (localStorage.getItem('walkthroughSeen')) {
      document.getElementById('walkthrough').style.display = 'none';
    }
    AOS.init({ duration: 800, easing: 'ease-in-out', once: true });
  }
</script>
